cmake_minimum_required(VERSION 3.16)
project(desktop_wave_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Desktop wave requires Windows")
endif()
add_executable(desktop_wave_probe main.cpp ../../windows/runner/desktop_wave.cpp)
target_compile_features(desktop_wave_probe PRIVATE cxx_std_17)
target_compile_definitions(desktop_wave_probe PRIVATE UNICODE _UNICODE)
target_include_directories(desktop_wave_probe PRIVATE ../../windows/runner)
target_link_libraries(desktop_wave_probe PRIVATE d3d11 dxgi d3dcompiler dcomp gdi32 ole32 user32)
if(MSVC)
  target_compile_options(desktop_wave_probe PRIVATE /W4 /WX /EHsc)
endif()
